  <action>
    In process_usings, add a glob branch as the FIRST check inside the `if let AstDecl::Using(using) = item` block, before the `path.len() == 1` check:

    ```rust
    // Glob import: `using Status::*;`
    // path ends with "*" (set by parser when ::* is written)
    if path.last().map(|s| s == "*").unwrap_or(false) {
        if path.len() < 2 {
            continue; // malformed, parser should have rejected
        }
        let enum_path = &path[..path.len() - 1];
        let enum_fqn = enum_path.join("::");

        // Prelude types (e.g., Option) are valid but vacuous -- None/Some already injected sub-prelude
        if prelude::PRELUDE_TYPE_NAMES.contains(&enum_fqn.as_str()) {
            continue;
        }

        if let Some(enum_def_id) = scope.def_map.get(&enum_fqn) {
            let entry = scope.def_map.get_entry(enum_def_id);
            if entry.kind != DefKind::Enum {
                diags.push(ResolutionError::UnresolvedName {
                    name: format!("{enum_fqn} (not an enum)"),
                    file: scope.current_file,
                    span: using.span,
                    suggestion: None,
                }.into());
                continue;
            }
            // Expand: one UsingEntry per variant
            // Enum variants are stored in namespace_members[enum_fqn] OR as FQNs "EnumName::VariantName" in by_fqn.
            // Try namespace_members first (enum name is the namespace for its variants in the collector).
            // If that is empty, fall back to iterating by_fqn with prefix matching.
            let variant_ids: Vec<_> = scope.def_map.pub_members_of(&enum_fqn).to_vec();
            if variant_ids.is_empty() {
                // Fallback: search by_fqn for keys starting with "{enum_fqn}::"
                for (fqn, &vid) in &scope.def_map.by_fqn {
                    if fqn.starts_with(&format!("{enum_fqn}::")) {
                        let ventry = scope.def_map.get_entry(vid);
                        scope.active_usings.push(UsingEntry {
                            alias: ventry.name.clone(),
                            target_ns: None,
                            target_fqn: Some(fqn.clone()),
                            span: using.span,
                            used: Cell::new(true),
                        });
                    }
                }
            } else {
                for variant_id in variant_ids {
                    let ventry = scope.def_map.get_entry(variant_id);
                    let target_fqn = if ventry.namespace.is_empty() {
                        format!("{}", ventry.name)
                    } else {
                        format!("{}::{}", ventry.namespace, ventry.name)
                    };
                    scope.active_usings.push(UsingEntry {
                        alias: ventry.name.clone(),
                        target_ns: None,
                        target_fqn: Some(target_fqn),
                        span: using.span,
                        used: Cell::new(true),
                    });
                }
            }
        } else {
            let suggestion = get_suggestion(&enum_fqn, scope);
            diags.push(ResolutionError::UnresolvedName {
                name: enum_fqn,
                file: scope.current_file,
                span: using.span,
                suggestion,
            }.into());
        }
        continue;
    }
    ```

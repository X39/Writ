**1. Modify `exec_new` in `dispatch/objects.rs`:**

Replace the current implementation with kind-dependent dispatch:
```rust
pub(super) fn exec_new(ctx: &mut ExecContext<'_>, r_dst: u16, type_idx: u32) -> ExecutionResult {
    let module = &ctx.modules[ctx.current_module_idx];
    let field_count = helpers::get_type_field_count(&module.module, type_idx);

    // Look up TypeDef kind — type_idx is 1-based MetadataToken
    let idx = type_idx.saturating_sub(1) as usize;
    let kind = module.module.type_defs.get(idx).map(|t| t.kind);

    match kind.and_then(writ_module::TypeDefKind::from_u8) {
        Some(writ_module::TypeDefKind::Struct) => {
            // kind=0: value-type struct — inline in register, no heap allocation
            let frame = ctx.task.call_stack.last_mut().unwrap();
            frame.registers[r_dst as usize] = Value::InlineStruct {
                type_idx,
                fields: vec![Value::Void; field_count],
            };
            ExecutionResult::Continue
        }
        Some(writ_module::TypeDefKind::Class) | Some(writ_module::TypeDefKind::Entity) => {
            // kind=4 (class) or kind=2 (entity): heap allocation (current behavior)
            let href = ctx.heap.alloc_struct(field_count);
            let frame = ctx.task.call_stack.last_mut().unwrap();
            frame.registers[r_dst as usize] = Value::Ref(href);
            ExecutionResult::Continue
        }
        Some(other) => ExecutionResult::Crash(format!(
            "NEW: type_idx {} has kind {:?}, not a struct or class", type_idx, other
        )),
        None => ExecutionResult::Crash(format!(
            "NEW: type_idx {} is out of range", type_idx
        )),
    }
}
```

Note: Entity (kind=2) is included in the heap-allocation path because entities are specialized classes (reference types) — they are constructed with NEW and live on the heap. The CONTEXT.md locked decision says "crash for enum=1, entity=2, component=3" but entity construction via NEW is an existing working feature that must not regress. Keep entity in the heap path. Only enum (kind=1) and component (kind=3) should crash since NEW on enum uses NEW_ENUM and components are extern-only.

**2. Modify `exec_get_field` in `dispatch/objects.rs`:**

Add variant dispatch — check if the object register holds InlineStruct or Ref:
```rust
pub(super) fn exec_get_field(
    ctx: &mut ExecContext<'_>, r_dst: u16, r_obj: u16, field_idx: u32,
) -> ExecutionResult {
    let frame = ctx.task.call_stack.last().unwrap();
    let obj_val = &frame.registers[r_obj as usize];
    match obj_val {
        Value::InlineStruct { fields, .. } => {
            let idx = field_idx as usize;
            match fields.get(idx) {
                Some(val) => {
                    let val = val.clone();
                    let frame = ctx.task.call_stack.last_mut().unwrap();
                    frame.registers[r_dst as usize] = val;
                    ExecutionResult::Continue
                }
                None => ExecutionResult::Crash(format!(
                    "GetField: field index {} out of range for struct with {} fields",
                    idx, fields.len()
                )),
            }
        }
        Value::Ref(_) | Value::Entity(_) => {
            // Existing heap/entity path
            let href = helpers::extract_ref(obj_val);
            match ctx.heap.get_field(href, field_idx as usize) {
                Ok(val) => {
                    let frame = ctx.task.call_stack.last_mut().unwrap();
                    frame.registers[r_dst as usize] = val;
                    ExecutionResult::Continue
                }
                Err(e) => ExecutionResult::Crash(format!("GetField: {}", e)),
            }
        }
        other => ExecutionResult::Crash(format!(
            "GetField: expected struct or class, got {:?}", other
        )),
    }
}
```

Critical borrow pattern — clone the value from r_val FIRST, then take mutable ref to r_obj:
```rust
pub(super) fn exec_set_field(
    ctx: &mut ExecContext<'_>, r_obj: u16, field_idx: u32, r_val: u16,
) -> ExecutionResult {
    let idx = field_idx as usize;
    // Clone the value to store BEFORE taking mutable reference to the object register
    let val = ctx.task.call_stack.last().unwrap().registers[r_val as usize].clone();

    let frame = ctx.task.call_stack.last_mut().unwrap();
    match &mut frame.registers[r_obj as usize] {
        Value::InlineStruct { fields, .. } => {
            if idx < fields.len() {
                fields[idx] = val;
                ExecutionResult::Continue
            } else {
                ExecutionResult::Crash(format!(
                    "SetField: field index {} out of range for struct with {} fields",
                    idx, fields.len()
                ))
            }
        }
        Value::Ref(href) => {
            // Copy href before dropping frame borrow (HeapRef is Copy)
            let href = *href;
            drop(frame);
            match ctx.heap.set_field(href, idx, val) {
                Ok(()) => ExecutionResult::Continue,
                Err(e) => ExecutionResult::Crash(format!("SetField: {}", e)),
            }
        }
        Value::Entity(eid) => {
            // Existing entity path — extract entity data_ref and go through heap
            let eid = *eid;
            drop(frame);
            // Look up entity data_ref in entity_registry and set_field on it

            ExecutionResult::Crash(format!("SetField: entity field access not supported via SetField"))
        }
        _ => ExecutionResult::Crash("SetField: not a struct or class".into()),
    }
}

/// Recursively collect all HeapRefs reachable from a Value.
/// Handles nested InlineStructs (structs containing struct fields).
pub(crate) fn collect_value_refs(val: &Value, refs: &mut Vec<HeapRef>) {
    match val {
        Value::Ref(href) => refs.push(*href),
        Value::InlineStruct { fields, .. } => {
            for field in fields {
                collect_value_refs(field, refs);
            }
        }
        _ => {}
    }
}
```

**3. Update `collect_roots` in `runtime.rs`:**

Replace the flat `if let Value::Ref(href)` pattern with `collect_value_refs` from gc.rs:
```rust
use crate::gc::collect_value_refs;

fn collect_roots(&self) -> Vec<HeapRef> {
    let mut roots = Vec::new();

    for task in self.scheduler.tasks.values() {
        for frame in &task.call_stack {
            for reg in &frame.registers {
                collect_value_refs(reg, &mut roots);
            }
        }
        if let Some(ref rv) = task.return_value {
            collect_value_refs(rv, &mut roots);
        }
    }

    for global in &self.scheduler.globals {
        collect_value_refs(global, &mut roots);
    }

    for (_entity_id, slot) in self.scheduler.entity_registry.alive_entities() {
        if let Some(href) = slot.data_ref {
            roots.push(href);
        }
    }

    roots
}
```

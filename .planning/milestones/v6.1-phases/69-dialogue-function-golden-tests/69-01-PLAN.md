```writ
// Dialogue/function interplay -- exercises dlg blocks calling fn helpers
// and fn calling dlg, with Tier 1 speaker params, code escapes, and transitions.
// No $ choice in this test (avoids known ::choice serialization bug).

// Explicit extern declarations required: dialogue lowering emits
// say(speaker, text) with 2 args, but auto-injected say has 1 param.
pub extern fn say(speaker: Entity, text: string);
pub extern fn say_localized(speaker: Entity, key: string, fallback: string);

// Helper function called from dialogue blocks
fn format_greeting(name: string) -> string {
    "Welcome, " + name + "!"
}

// Helper function that computes a value
fn quest_reward(base: int, multiplier: int) -> int {
    base * multiplier
}

// A dialogue block that calls helper functions via $ code escape
dlg greet(npc: Entity, player_name: string) {
    @npc Hello, traveler.
    $ let greeting: string = format_greeting(player_name);
    @npc {greeting}
    $ let reward: int = quest_reward(10, 3);
    $ ::log::info("Computed reward");
}

// A dialogue block that transitions to another dialogue
dlg intro(narrator: Entity) {
    @narrator The adventure begins here.
    @narrator Are you ready?
    -> quest_start(narrator)
}

// Target of the transition
dlg quest_start(narrator: Entity) {
    @narrator Your quest has begun!
    $ let progress: int = 0;
    @narrator Good luck on your journey.
}

// A regular function that calls a dialogue function
fn start_conversation(npc: Entity) {
    greet(npc, "Hero");
}

fn main() {
    ::log::info("Starting dialogue test");
}
```

```writ
// Full quest pattern -- entity + dialogue + functions + enum + choices.
// Exercises the complete game scripting pattern: entity declaration,
// dialogue trees with speakers, helper functions, enum-based state, and choices.
//
// NOTE: $ choice uses ::choice/::ChoiceOption which lowers to fn() {} lambdas.
// If this triggers the known serialization bug (UnexpectedEof in multi-function
// modules), remove the $ choice block and document the limitation.

// Explicit extern declarations for dialogue speaker resolution.
pub extern fn say(speaker: Entity, text: string);
pub extern fn say_localized(speaker: Entity, key: string, fallback: string);

enum QuestState {
    NotStarted,
    Active,
    Completed,
}

entity Merchant {
    name: string,
    has_quest: bool,
}

// Check if a quest can be offered
fn can_offer_quest(state: QuestState) -> bool {
    match state {
        QuestState::NotStarted => { true }
        QuestState::Active => { false }
        QuestState::Completed => { false }
    }
}

// Calculate quest reward based on state
fn get_reward(base: int) -> int {
    base * 2
}

// Opening dialogue with the merchant
dlg merchant_greeting(merchant: Entity) {
    @merchant Welcome to my shop, traveler!
    @merchant I have something for you.
    -> merchant_quest(merchant)
}

// Quest offer dialogue -- exercises $ if conditional
dlg merchant_quest(merchant: Entity) {
    $ let state: QuestState = QuestState::NotStarted;
    $ let available: bool = can_offer_quest(state);
    $ if available {
        @merchant I have a task for you.
        @merchant Retrieve the ancient scroll from the ruins.
    } else {
        @merchant You already have a task. Come back later.
    }
}

// Quest completion dialogue with reward calculation
dlg merchant_complete(merchant: Entity) {
    @merchant You found the scroll! Excellent work.
    $ let reward: int = get_reward(50);
    @merchant Here is your reward.
    $ ::log::info("Quest completed");
}

fn main() {
    ::log::info("Starting quest pattern test");
}
```

IMPORTANT on the `$ choice` question: The CONTEXT.md identifies a known bug where `::choice` with `fn() {}` lambdas causes UnexpectedEof in multi-function modules. Since `dlg_quest_pattern.writ` has multiple function/dialogue declarations, `$ choice` will almost certainly trigger this bug. Per the user decision, use `$ if` conditional instead of `$ choice` in this test file to avoid the bug. The `$ if` conditional exercises the dialogue conditional path without hitting the serialization issue. If the executor wants to try `$ choice`, they should attempt it first and fall back to the version above if serialization fails.

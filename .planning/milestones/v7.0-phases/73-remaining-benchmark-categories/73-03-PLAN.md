**oop_dispatch.writ (contract dispatch version):**
```writ
contract Computable {
    fn compute(self) -> int;
}

pub class TypeA {}
pub class TypeB {}
pub class TypeC {}
pub class TypeD {}

impl Computable for TypeA {
    fn compute(self) -> int { 1 }
}

impl Computable for TypeB {
    fn compute(self) -> int { 2 }
}

impl Computable for TypeC {
    fn compute(self) -> int { 3 }
}

impl Computable for TypeD {
    fn compute(self) -> int { 4 }
}

fn main() {
    let mut sum: int = 0;
    let mut i: int = 0;
    while i < 100000 {
        let r: int = i % 4;
        if r == 0 {
            let obj: TypeA = new TypeA {};
            sum = sum + obj.compute();
        }
        if r == 1 {
            let obj: TypeB = new TypeB {};
            sum = sum + obj.compute();
        }
        if r == 2 {
            let obj: TypeC = new TypeC {};
            sum = sum + obj.compute();
        }
        if r == 3 {
            let obj: TypeD = new TypeD {};
            sum = sum + obj.compute();
        }
        i = i + 1;
    }
    log::info($"{sum}");
}
```
Note: Writ does not have `else if` chains or dynamic dispatch via a base-type variable in the current compiler. Each branch creates a concrete type and calls compute() on it. The CALL_VIRT instruction is still emitted because the method is defined via `impl Computable for TypeX`. If the compiler does NOT use CALL_VIRT for this pattern (because the concrete type is known), that is fine -- the benchmark still measures function call overhead. The key is algorithmic equivalence: same 4 types, same cycling, same sum.

```rust
trait Computable {
    fn compute(&self) -> i64;
}

struct TypeA;
struct TypeB;
struct TypeC;
struct TypeD;

impl Computable for TypeA { fn compute(&self) -> i64 { 1 } }
impl Computable for TypeB { fn compute(&self) -> i64 { 2 } }
impl Computable for TypeC { fn compute(&self) -> i64 { 3 } }
impl Computable for TypeD { fn compute(&self) -> i64 { 4 } }

fn main() {
    let mut objects: Vec<Box<dyn Computable>> = Vec::with_capacity(100_000);
    for i in 0..100_000i64 {
        let obj: Box<dyn Computable> = match i % 4 {
            0 => Box::new(TypeA),
            1 => Box::new(TypeB),
            2 => Box::new(TypeC),
            3 => Box::new(TypeD),
            _ => unreachable!(),
        };
        objects.push(obj);
    }
    let sum: i64 = objects.iter().map(|o| o.compute()).sum();
    println!("{}", sum);
}
```

**Actually, the simplest approach**: Add a new test to `writ-golden/tests/golden_tests.rs` that:
1. Defines a library source string with `pub class Stub { value: int }`
2. Defines a user source string with `fn main() { let s: Stub = new Stub { value: 1 }; }`
3. Compiles BOTH sources together through the pipeline using `writ_compiler::resolve::resolve` with two file entries (FileId(0) for lib, FileId(1) for user)
4. Asserts that resolution, type-checking, and emission all succeed

This validates the cross-file resolution path that `with_library` relies on at the IL/runtime level. The runtime-level `with_library` test will come in Plan 03.

Add the test:
```rust
#[test]
fn lib_preload_cross_file_resolution() {
    // Library source
    let lib_src: &'static str = Box::leak("pub class Stub { value: int }".to_string().into_boxed_str());
    // User source that references the library type
    let user_src: &'static str = Box::leak("fn main() {\n    let s: Stub = new Stub { value: 1 };\n}".to_string().into_boxed_str());

    let handle = std::thread::Builder::new()
        .stack_size(16 * 1024 * 1024)
        .spawn(move || -> Result<(), String> {
            use writ_diagnostics::{FileId, Severity};

            let lib_fid = FileId(0);
            let user_fid = FileId(1);

            // Parse + lower both
            let (lib_cst, lib_errs) = writ_parser::parse(lib_src);
            assert!(lib_errs.is_empty(), "lib parse errors: {:?}", lib_errs);
            let (lib_ast, lib_lower_errs) = writ_compiler::lower(lib_cst.unwrap());
            assert!(lib_lower_errs.is_empty());

            let (user_cst, user_errs) = writ_parser::parse(user_src);
            assert!(user_errs.is_empty(), "user parse errors: {:?}", user_errs);
            let (user_ast, user_lower_errs) = writ_compiler::lower(user_cst.unwrap());
            assert!(user_lower_errs.is_empty());

            // Resolve together
            let asts = vec![(lib_fid, &lib_ast), (user_fid, &user_ast)];
            let paths = vec![(lib_fid, "lib.writ"), (user_fid, "user.writ")];
            let (resolved, resolve_diags) = writ_compiler::resolve::resolve(&asts, &paths);
            let has_errors = resolve_diags.iter().any(|d| d.severity == Severity::Error);
            assert!(!has_errors, "resolution errors: {:?}", resolve_diags.iter().map(|d| &d.message).collect::<Vec<_>>());

            // Typecheck
            let (_typed, _interner, _env, type_diags) = writ_compiler::check::typecheck(resolved, &asts);
            let has_type_errors = type_diags.iter().any(|d| d.severity == Severity::Error);
            assert!(!has_type_errors, "type errors: {:?}", type_diags.iter().map(|d| &d.message).collect::<Vec<_>>());

            Ok(())
        })
        .unwrap()
        .join()
        .unwrap();

    handle.unwrap();
}
```

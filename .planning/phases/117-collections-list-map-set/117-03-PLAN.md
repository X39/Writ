**Step 1: Create writ-cli/build.rs**

This build script compiles `writ-std/src/collections.writ` at build time using `writ_compiler::compile_source` and writes the output to `$OUT_DIR/writ-std.writc`.

```rust
//! Build script: compiles writ-std source into a .writc binary for embedding.

fn main() {
    // Re-run if the writ-std source changes
    println!("cargo:rerun-if-changed=../writ-std/src/collections.writ");

    let src = include_str!("../writ-std/src/collections.writ");
    // compile_source requires &'static str
    let src_static: &'static str = Box::leak(src.to_string().into_boxed_str());

    // Compile on a 16MB stack thread (deep AST recursion)
    let handle = std::thread::Builder::new()
        .stack_size(16 * 1024 * 1024)
        .spawn(move || {
            writ_compiler::compile_source(src_static)
        })
        .expect("failed to spawn compile thread");

    let bytes = handle.join()
        .expect("compile thread panicked")
        .expect("writ-std compilation failed");

    let out_dir = std::env::var("OUT_DIR").unwrap();
    let out_path = std::path::Path::new(&out_dir).join("writ-std.writc");
    std::fs::write(&out_path, &bytes).expect("failed to write writ-std.writc");
}
```

**Helper functions:**

```rust
//! Integration tests: compile and execute writ-std collections at runtime.

const WRIT_STD_SRC: &str = include_str!("../../writ-std/src/collections.writ");

fn compile(src: &str) -> Vec<u8> {
    let src_static: &'static str = Box::leak(src.to_string().into_boxed_str());
    std::thread::Builder::new()
        .stack_size(16 * 1024 * 1024)
        .spawn(move || writ_compiler::compile_source(src_static).unwrap())
        .unwrap()
        .join()
        .unwrap()
}

fn find_main_idx(module: &writ_module::Module) -> usize {
    module.method_defs.iter().enumerate()
        .find(|(_, md)| {
            writ_module::heap::read_string(&module.string_heap, md.name)
                .unwrap_or("") == "main"
        })
        .map(|(idx, _)| idx)
        .expect("no main method found")
}

/// Run user_src concatenated with writ-std, expect completion without panic.
fn run_to_completion(user_src: &str) {
    let full_src = format!("{}\n\n{}", WRIT_STD_SRC, user_src);
    let bytes = compile(&full_src);
    let module = writ_module::Module::from_bytes(&bytes).unwrap();
    let main_idx = find_main_idx(&module);
    let mut runtime = writ_runtime::RuntimeBuilder::new(module)
        .with_gc()
        .build()
        .unwrap();
    runtime.spawn_task(main_idx, vec![]).unwrap();
    match runtime.tick(0.0, writ_runtime::ExecutionLimit::None) {
        writ_runtime::TickResult::AllCompleted | writ_runtime::TickResult::Empty => {}
        other => panic!("unexpected tick result: {:?}", other),
    }
}

/// Run user_src with writ-std loaded as a SEPARATE library module via with_library().
/// This tests the cross-module loading path used by writ-cli.
fn run_with_library(user_src: &str) {
    let std_bytes = compile(WRIT_STD_SRC);
    let std_module = writ_module::Module::from_bytes(&std_bytes).unwrap();
    let user_bytes = compile(user_src);
    let user_module = writ_module::Module::from_bytes(&user_bytes).unwrap();
    let main_idx = find_main_idx(&user_module);
    let mut runtime = writ_runtime::RuntimeBuilder::new(user_module)
        .with_library(std_module)
        .with_gc()
        .build()
        .unwrap();
    runtime.spawn_task(main_idx, vec![]).unwrap();
    match runtime.tick(0.0, writ_runtime::ExecutionLimit::None) {
        writ_runtime::TickResult::AllCompleted | writ_runtime::TickResult::Empty => {}
        other => panic!("unexpected tick result: {:?}", other),
    }
}
```

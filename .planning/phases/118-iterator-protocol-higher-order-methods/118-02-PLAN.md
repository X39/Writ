```rust
} else if let TyKind::Class(class_def_id) = ctx.interner.kind(typed_iterable.ty()).clone() {
    // Look up Iterable<T> impl for this class
    let iterable_name = "Iterable";
    let iterator_name = "Iterator";

    // Find the Iterable contract DefId from the def_map
    let iterable_contract_def_id = ctx.def_map.get(iterable_name);
    let iterator_contract_def_id = ctx.def_map.get(iterator_name);

    if let (Some(iterable_did), Some(iterator_did)) = (iterable_contract_def_id, iterator_contract_def_id) {
        // Check if class implements Iterable<T> via impl_index
        let has_iterable = ctx.type_env.impl_index.get(&class_def_id)
            .map(|impls| impls.iter().any(|entry| entry.contract_def_id == Some(iterable_did)))
            .unwrap_or(false);

        if has_iterable {
            // Find the iterator() method's return type from the Iterable impl
            // The return type is Iterator<T> (a contract type)
            // The next() method on Iterator<T> returns T? (Option<T>)
            // We need to extract T from Option<T> to get elem_ty

            // Find the Iterable impl entry
            let iterable_impl = ctx.type_env.impl_index.get(&class_def_id).unwrap()
                .iter()
                .find(|e| e.contract_def_id == Some(iterable_did))
                .unwrap();

            // Get the iterator() method signature
            let iterator_method_sig = iterable_impl.methods.iter()
                .find(|(name, _)| name == "iterator")
                .map(|(_, sig)| sig);

            if let Some(sig) = iterator_method_sig {
                // sig.return_type is Iterator<T> (TyKind::Contract)
                // Look up Iterator<T>.next() return type from contract_methods
                let next_ret_ty = ctx.type_env.contract_methods.get(&iterator_did)
                    .and_then(|methods| methods.iter().find(|m| m.name == "next"))
                    .map(|m| m.return_type);

                if let Some(next_ty) = next_ret_ty {
                    // next_ty is T? (Option<T>) — unwrap the Option to get T
                    if let TyKind::Option(inner) = ctx.interner.kind(next_ty).clone() {
                        inner  // This is elem_ty
                    } else {
                        // Fallback: next() doesn't return Option — use as-is
                        next_ty
                    }
                } else {
                    ctx.interner.error()
                }
            } else {
                ctx.interner.error()
            }
        } else {
            // Class doesn't implement Iterable — emit error
            ctx.interner.error()
        }
    } else {
        ctx.interner.error()
    }
} else {
    ctx.interner.error()
}
```

```rust
TyKind::Class(_class_def_id) => {
    // Iterator protocol desugaring:
    // 1. r_collection = emit iterable
    // 2. r_iter = CALL_VIRT r_collection, Iterable.iterator()
    // 3. loop:
    //    r_next = CALL_VIRT r_iter, Iterator.next()
    //    r_is_none = IS_NONE r_next
    //    BR_TRUE r_is_none, loop_end
    //    r_elem = UNWRAP r_next
    //    <bind r_elem to loop var>
    //    <body>
    //    BR loop_start
    // loop_end:

    let int_ty = crate::check::ty::Ty(0);
    let bool_ty = crate::check::ty::Ty(2);

    let r_collection = emit_expr(emitter, iterable);

    // Resolve Iterable contract token and iterator() slot
    let iterable_did = iterable_contract_def_id.unwrap();
    let iterator_did = iterator_contract_def_id.unwrap();

    let iterable_token = emitter.builder.token_for_def(iterable_did)
        .map(|t| t.0).unwrap_or(0);
    let iterator_slot = emitter.builder.contract_method_slot_by_name(iterable_did, "iterator")
        .unwrap_or(0);

    // Allocate register for the Iterator<T> object
    // The iterator's type is a contract type — use an opaque type for the register
    let iterator_ty = emitter.interner.intern(TyKind::Contract(iterator_did));
    let r_iter = emitter.alloc_reg(iterator_ty);

    // CALL_VIRT: r_iter = collection.iterator()
    // argc=1 (self only), r_base = r_collection
    let r_base_iter = pack_args_consecutive(emitter, &[r_collection]);
    emitter.emit(Instruction::CallVirt {
        r_dst: r_iter,
        r_obj: r_collection,
        contract_idx: iterable_token,
        slot: iterator_slot,
        r_base: r_base_iter,
        argc: 1,
    });

    // Resolve Iterator contract token and next() slot
    let iterator_token = emitter.builder.token_for_def(iterator_did)
        .map(|t| t.0).unwrap_or(0);
    let next_slot = emitter.builder.contract_method_slot_by_name(iterator_did, "next")
        .unwrap_or(0);

    // Allocate register for T? (Option<elem_ty>)
    let option_ty = emitter.interner.intern(TyKind::Option(binding_ty));
    let r_next = emitter.alloc_reg(option_ty);

    // Labels
    let loop_start = emitter.new_label();
    let loop_end = emitter.new_label();
    emitter.push_loop(loop_end, loop_start);

    emitter.mark_label_here(loop_start);

    // CALL_VIRT: r_next = _iter.next()
    // argc=1 (self only)
    let r_base_next = pack_args_consecutive(emitter, &[r_iter]);
    emitter.emit(Instruction::CallVirt {
        r_dst: r_next,
        r_obj: r_iter,
        contract_idx: iterator_token,
        slot: next_slot,
        r_base: r_base_next,
        argc: 1,
    });

    // IS_NONE: check if next() returned null
    let r_is_none = emitter.alloc_reg(bool_ty);
    emitter.emit(Instruction::IsNone { r_dst: r_is_none, r_opt: r_next });

    // BR_TRUE r_is_none, loop_end
    let brt_idx = emitter.instructions.len();
    emitter.emit(Instruction::BrTrue { r_cond: r_is_none, offset: 0 });
    emitter.add_fixup(brt_idx, loop_end);

    // UNWRAP: r_elem = unwrap r_next
    let r_elem = emitter.alloc_reg(binding_ty);
    emitter.emit(Instruction::Unwrap { r_dst: r_elem, r_opt: r_next });

    // Bind element to loop variable
    emitter.locals.insert(binding.to_string(), r_elem);

    // Emit body
    for stmt in body {
        emit_stmt(emitter, stmt);
    }

    // BR loop_start
    let br_idx = emitter.instructions.len();
    emitter.emit(Instruction::Br { offset: 0 });
    emitter.add_fixup(br_idx, loop_start);

    emitter.mark_label_here(loop_end);
    emitter.pop_loop();
}
_ => {
    // Unknown iterable type — emit Nop (should not be reached after check_stmt)
    let _ = emit_expr(emitter, iterable);
    emitter.emit(Instruction::Nop);
}
```

```writ
pub class ListIterator<T> {
    source: T[],
    index: int

impl<T> ListIterator<T> {
    pub fn next(mut self) -> T? {
        if self.index >= self.source.len() {
            return null;
        }
        let item: T = self.source[self.index];
        self.index = self.index + 1;
        item
    }
}

impl<T> Iterator<T> for ListIterator<T> {
    pub fn next(mut self) -> T? {
        if self.index >= self.source.len() {
            return null;
        }
        let item: T = self.source[self.index];
        self.index = self.index + 1;
        item
    }
}

pub class List<T> {
    items: T[]
}

impl<T> List<T> {
    pub fn add(mut self, item: T) {
        self.items.add(item);
    }

    pub fn len(self) -> int {
        self.items.len()
    }
}

impl<T> Iterable<T> for List<T> {
    pub fn iterator(self) -> Iterator<T> {
        new ListIterator<T> { source: self.items, index: 0 }

    for x in list {
        sum = sum + x;
    }
    let _result: int = sum;
}
```

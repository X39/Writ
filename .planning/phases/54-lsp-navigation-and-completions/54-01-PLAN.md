```rust
/// Format a type as a human-readable string, resolving named types via DefMap.
///
/// Unlike `display()` which shows "struct"/"entity"/etc., this shows the actual
/// type name (e.g., "Potion", "Player") by looking up DefEntry::name.
pub fn display_named(&self, ty: Ty, def_map: &crate::resolve::def_map::DefMap) -> String {
    match self.kind(ty) {
        TyKind::Struct(def_id) | TyKind::Class(def_id)
        | TyKind::Entity(def_id) | TyKind::Enum(def_id) => {
            def_map.get_entry(*def_id).name.clone()
        }
        TyKind::Array(elem) => format!("{}[]", self.display_named(*elem, def_map)),
        TyKind::Option(inner) => format!("Option<{}>", self.display_named(*inner, def_map)),
        TyKind::Result(ok, err) => format!(
            "Result<{}, {}>",
            self.display_named(*ok, def_map),
            self.display_named(*err, def_map)
        ),
        TyKind::Func { params, ret } => {
            let ps: Vec<String> = params.iter()
                .map(|p| self.display_named(*p, def_map)).collect();
            format!("fn({}) -> {}", ps.join(", "), self.display_named(*ret, def_map))
        }
        TyKind::TaskHandle(inner) => format!("TaskHandle<{}>", self.display_named(*inner, def_map)),
        _ => self.display(ty), // primitives, GenericParam, Infer, Error
    }
}
```

**Function 1: `position_to_byte_offset`** -- Convert LSP Position (0-based line, UTF-16 character) to a byte offset. This is the inverse of `convert::offset_to_position`.

```rust
pub fn position_to_byte_offset(source: &str, pos: lsp_types::Position) -> Option<usize> {
    let mut current_line: u32 = 0;
    let mut byte_idx: usize = 0;
    let mut iter = source.char_indices().peekable();

    // Skip past pos.line newlines
    while current_line < pos.line {
        match iter.next() {
            Some((_, '\n')) => current_line += 1,
            None => return None,
            _ => {}
        }
    }

    // Now advance pos.character UTF-16 code units within the current line
    let mut utf16_col: u32 = 0;
    while utf16_col < pos.character {
        match iter.next() {
            Some((_, '\n')) | None => return None, // past end of line
            Some((_, ch)) => utf16_col += ch.len_utf16() as u32,
        }
    }

    // The current position in the iterator is our byte offset
    match iter.peek() {
        Some(&(idx, _)) => Some(idx),
        None => Some(source.len()), // end of file
    }
}
```

**Function 2: `expr_at_offset`** -- Walk a TypedAst to find the innermost TypedExpr whose span contains a given byte offset.

Walk ALL body-containing decl variants: `Fn { body }`, `Impl { methods }` (each method body), `Const { value }`, `Global { value }`. For each body, recursively walk to find the narrowest-span expression containing the offset.

The walker function `find_in_expr(expr: &TypedExpr, offset: usize) -> Option<&TypedExpr>` must:
- Check if `expr.span().start <= offset && offset < expr.span().end`
- If yes, recurse into sub-expressions (children) to find a narrower match
- Return the narrowest matching expression
- Handle ALL TypedExpr variants that contain sub-expressions: Call (callee + args), Field (receiver), ComponentAccess (receiver), Index (receiver + index), Binary (left + right), UnaryPrefix (expr), Match (scrutinee + arm bodies), If (condition + then + else), Block (stmts + tail), Lambda (body), Assign (target + value), New (field values), ArrayLit (elements), Range (start + end), Spawn/SpawnDetached/Join/Cancel/Defer (expr), Return (value)
- Also walk into TypedStmt for Block contents: Let (value), Expr (expr), For (iterable + body stmts), While (condition + body stmts), Atomic (body stmts), Return (value), Break (value)

**Function 3: `find_def_id_at_offset`** -- Given a TypedExpr, extract the DefId it references (if any).

```rust
pub fn find_def_id_at_offset(expr: &TypedExpr, def_map: &DefMap) -> Option<DefId> {
    match expr {
        TypedExpr::Call { callee_def_id: Some(id), .. } => Some(*id),
        TypedExpr::Var { name, .. } => {
            // Look up in DefMap by_fqn or by iterating arena for matching name
            def_map.by_fqn.values()
                .find(|&&id| def_map.get_entry(id).name == *name)
                .copied()
                // Also check file_private maps
                .or_else(|| {
                    for privs in def_map.file_private.values() {
                        if let Some(&id) = privs.get(name) {
                            return Some(id);
                        }
                    }
                    None
                })
        }
        TypedExpr::New { target_def_id, .. } => Some(*target_def_id),
        TypedExpr::Path { segments, .. } => {
            let fqn = segments.join("::");
            def_map.get(&fqn)
        }
        _ => None,
    }
}
```

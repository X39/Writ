**Step 1: Add `contract_method_slot_by_name` to ModuleBuilder** (writ-compiler/src/emit/module_builder.rs)

Add this public method after the existing `contract_token_for_method_def_id` method (around line 1070):

```rust
/// Look up the CALL_VIRT slot for a contract method by contract DefId and method name.
///
/// Searches the ContractMethod entries for the ContractDef registered with `contract_def_id`,
/// returning the 0-based slot index of the entry whose name matches `method_name`.
/// Slots are assigned by `assign_vtable_slots` in declaration order (0, 1, 2, ...).
///
/// Returns None if the contract is not registered or the method is not found.
pub fn contract_method_slot_by_name(&self, contract_def_id: DefId, method_name: &str) -> Option<u16> {
    let contract_idx = self.contract_def_def_ids
        .iter()
        .position(|id| id.as_ref() == Some(&contract_def_id))?;
    let range = self.contract_method_range(contract_idx);
    for cm_idx in range {
        let name_in_heap = self.string_heap.get_str(self.contract_methods[cm_idx].row.name);
        if name_in_heap == method_name {
            return Some(self.contract_methods[cm_idx].row.slot);
        }
    }
    None
}
```

**Step 2: Add TyKind::Contract branch in emit_expr** (writ-compiler/src/emit/body/expr/mod.rs)

Inside the `TypedExpr::Call` arm, BEFORE the `if !is_static_call && matches!(...)` check (line 274), add a new block that detects contract-typed receivers and emits CALL_VIRT immediately:

```rust
// Contract-typed receiver dispatch (EMIT-01, EMIT-02)
// Must come BEFORE Branch A's !is_static_call check because:
// - callee_def_id is None for contract method calls (Branch A intercepts)
// - callee type is Func (Branch A's matches! check passes)
// - extract_type_def_id returns None for TyKind::Contract (falls to CALL_INDIRECT)
if !is_static_call {
    if let TypedExpr::Field { receiver, field, .. } = callee.as_ref() {
        if let TyKind::Contract(contract_def_id) = emitter.interner.kind(receiver.ty()).clone() {
            let TypedExpr::Call { ty, args, .. } = expr else { unreachable!() };
            let r_dst_call = emitter.alloc_reg(*ty);

            // Emit self (receiver) first, then remaining args
            let r_self = emit_expr(emitter, receiver);
            let arg_regs: Vec<u16> = std::iter::once(r_self)
                .chain(args.iter().map(|arg| emit_expr(emitter, arg)))
                .collect();
            let r_base = pack_args_consecutive(emitter, &arg_regs);

            // Resolve contract token and slot by name (not by callee_def_id — it's None)
            let contract_token = emitter.builder.token_for_def(contract_def_id)
                .map(|t| t.0)
                .unwrap_or(0);
            let slot = emitter.builder.contract_method_slot_by_name(contract_def_id, field)
                .unwrap_or(0);

            // CALL_VIRT layout: r_obj = receiver, remaining args after
            let r_obj = r_base;
            let r_args_base = if arg_regs.len() > 1 { r_base + 1 } else { r_base };
            let n_args = (arg_regs.len() as u16).saturating_sub(1);
            emitter.emit(Instruction::CallVirt {
                r_dst: r_dst_call,
                r_obj,
                contract_idx: contract_token,
                slot,
                r_base: r_args_base,
                argc: n_args,
            });
            return r_dst_call;
        }
    }
}
```

/// Helper: build a ModuleBuilder with a ContractDef that has a DefId, plus methods and an impl.
/// Returns (builder, contract_def_id, contract_token).
fn make_builder_with_contract_receiver(
    contract_def_id: DefId,
    struct_def_id: DefId,
) -> ModuleBuilder {
    let mut builder = ModuleBuilder::new();

    // ContractDef with DefId so token_for_def works
    let contract_handle = builder.add_contract_def("MyContract", "test", Some(contract_def_id));
    builder.add_contract_method(contract_handle, "first_method", 0, 0);
    builder.add_contract_method(contract_handle, "second_method", 0, 0);

    // TypeDef for the implementing struct
    let type_handle = builder.add_typedef("MyClass", "test", TypeDefKind::Class, 0, Some(struct_def_id));

    // MethodDef for the impl methods
    let (_, impl_method_def_id_1) = make_def_id();
    builder.add_methoddef(Some(type_handle), "first_method", 0, 0, Some(impl_method_def_id_1), 0);
    let (_, impl_method_def_id_2) = make_def_id();
    builder.add_methoddef(Some(type_handle), "second_method", 0, 0, Some(impl_method_def_id_2), 0);

    // Assign vtable slots before finalize
    slots::assign_vtable_slots(&mut builder);
    builder.finalize();

    builder
}

fn test_contract_receiver_emits_call_virt() {

    let mut interner = make_interner();

        ret: Box::new(ty_void),

        callee_def_id: None,

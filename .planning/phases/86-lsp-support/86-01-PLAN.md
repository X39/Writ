    ```rust
    #[tokio::test(flavor = "multi_thread", worker_threads = 2)]
    async fn test_diagnostics_contract_valid_no_errors() {
        let mut client = LspClient::start().await;
        let source = r#"contract Speaker {
        fn speak(self, msg: string) -> void;
    }
    pub struct NPC impl Speaker {
        fn speak(self, msg: string) -> void {}
    }
    pub fn main() {
        let s: Speaker = new NPC {};
        s.speak("hello");
    }
    "#;
        let uri = "file:///test/contract_valid.writ";
        let diag_notifications = client.open_document_and_collect_diagnostics(uri, source).await;
        // Collect diagnostics for our URI
        let mut all_diagnostics: Vec<serde_json::Value> = Vec::new();
        for notif in &diag_notifications {
            let params = notif.get("params").cloned().unwrap_or(json!({}));
            let notif_uri = params.get("uri").and_then(|v| v.as_str()).unwrap_or("");
            if notif_uri == uri {
                if let Some(diags) = params.get("diagnostics").and_then(|v| v.as_array()) {
                    all_diagnostics.extend(diags.clone());
                }
            }
        }
        assert!(all_diagnostics.is_empty(),
            "valid contract code should produce zero diagnostics, got: {:?}", all_diagnostics);
        client.shutdown().await;
    }
    ```

        let uri = "file:///test/contract_invalid.writ";

        let mut all_diagnostics: Vec<serde_json::Value> = Vec::new();
        for notif in &diag_notifications {
            let params = notif.get("params").cloned().unwrap_or(json!({}));
            let notif_uri = params.get("uri").and_then(|v| v.as_str()).unwrap_or("");
            if notif_uri == uri {

        });

        client.shutdown().await;
    }

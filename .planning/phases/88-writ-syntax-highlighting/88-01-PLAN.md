    The script logic:

    ```python
    import sys, os

    def update_fences(path):
        with open(path, 'r', encoding='utf-8') as f:
            lines = f.readlines()
        result = []
        in_block = False
        changed = 0
        for line in lines:
            stripped = line.rstrip('\n').rstrip('\r')
            if not in_block and stripped == '```':
                # Bare opening fence -> tag as writ
                result.append('```writ\n')
                in_block = True
                changed += 1
            elif in_block and stripped == '```':
                # Closing fence -> leave as-is
                result.append(line)
                in_block = False
            elif not in_block and stripped.startswith('```'):
                # Already-tagged opening fence (```toml, ```csv, ```ebnf, ```writ, etc.)
                result.append(line)
                in_block = True
            else:
                result.append(line)
        with open(path, 'w', encoding='utf-8', newline='\n') as f:
            f.writelines(result)
        return changed

    if __name__ == '__main__':
        total = 0
        for path in sys.argv[1:]:
            n = update_fences(path)
            if n > 0:
                print(f"  {path}: {n} fences updated")
            total += n
        print(f"Total: {total} fences updated")
    ```

    ```rust
    fn collect_attribute_decl_defs(
        typed_ast: &TypedAst,
        asts: &[(FileId, &Ast)],
        builder: &mut ModuleBuilder,
    ) {
        let def_map = &typed_ast.def_map;
        for decl in &typed_ast.decls {
            let def_id = match decl {
                TypedDecl::AttributeDef { def_id } => *def_id,
                _ => continue,
            };
            let entry = def_map.get_entry(def_id);

            // Find the AstAttributeDecl to get param types
            let mut params: &[AstParam] = &[];
            'outer: for (_, ast) in asts {
                for d in &ast.items {
                    if let AstDecl::Attribute(a) = d {
                        if a.name == entry.name && a.name_span == entry.name_span {
                            params = &a.params;
                            break 'outer;
                        }
                    }
                }
            }

            // Encode param type signature as blob: u16 count, then each param's type tag
            // Use a simple encoding: for each param, write the type as a tag byte
            // (matching AttrValue tags: 0x01=string, 0x02=int, 0x03=bool)
            let mut sig_buf = Vec::new();
            sig_buf.extend_from_slice(&(params.len() as u16).to_le_bytes());
            for param in params {
                let tag = match &param.ty {
                    AstType::Named { name, .. } => match name.as_str() {
                        "string" => writ_module::attr::ATTR_TAG_STRING,
                        "int" => writ_module::attr::ATTR_TAG_INT,
                        "bool" => writ_module::attr::ATTR_TAG_BOOL,
                        _ => continue, // unsupported type, already caught by checker
                    },
                    _ => continue,
                };
                sig_buf.push(tag);
            }

            let blob_offset = if sig_buf.is_empty() {
                0
            } else {
                builder.blob_heap.intern(&sig_buf)
            };

            builder.add_attribute_def(
                MetadataToken::new(0, 0), // NULL owner -- this is a declaration, not an application
                writ_module::tables::ATTR_OWNER_KIND_DECL,
                &entry.name,
                blob_offset,
            );
        }
    }
    ```

1. **collect_defs filtering** (`writ-compiler/src/emit/collect/mod.rs`):
   Add `active_conditions: &std::collections::HashSet<String>` parameter to `collect_defs`.

   Before the existing decl iteration loop, add a pre-scan that computes `skipped_def_ids: HashSet<DefId>`:
   ```rust
   let mut skipped_def_ids: std::collections::HashSet<DefId> = std::collections::HashSet::default();
   // Track which fallbacks have an active conditional pointing at them
   let mut active_for_fallback: FxHashMap<DefId, Vec<DefId>> = FxHashMap::default();

   for (&cond_def_id, cond_name) in &typed_ast.conditional_fns {
       let is_active = active_conditions.contains(cond_name.as_str());
       if is_active {
           // Active condition: skip the fallback, emit the conditional variant
           if let Some(&fb_id) = typed_ast.fallback_for_conditional.get(&cond_def_id) {
               skipped_def_ids.insert(fb_id);
               active_for_fallback.entry(fb_id).or_default().push(cond_def_id);
           }
       } else {
           // Inactive condition: skip the conditional variant, emit fallback
           skipped_def_ids.insert(cond_def_id);
       }
   }

   // E0010: check for ambiguous active conditions (multiple active conditionals sharing same fallback)
   for (fb_id, active_conds) in &active_for_fallback {
       if active_conds.len() > 1 {
           let fb_entry = def_map.get_entry(*fb_id);
           let cond_names: Vec<&str> = active_conds.iter()
               .map(|id| typed_ast.conditional_fns[id].as_str())
               .collect();
           diags.push(
               Diagnostic::error()
                   .with_code(writ_diagnostics::code::E0010)
                   .with_message(format!(
                       "multiple active conditions match function '{}': {}",
                       fb_entry.name,
                       cond_names.join(", ")
                   ))
                   .with_labels(vec![Label::primary(fb_entry.file_id, fb_entry.name_span)])
           );
       }
   }
   ```

   In the main decl iteration loop, for the `TypedDecl::Fn { def_id, .. }` arm, add at the top:
   ```rust
   if skipped_def_ids.contains(def_id) {
       continue;
   }
   ```

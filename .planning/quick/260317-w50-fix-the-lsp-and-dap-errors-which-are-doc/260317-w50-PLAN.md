**PART 3: New `hover_text_for_def` function in queries.rs (Bugs 4, 5, 7)**

Add a new public function for hover on declaration names:

```rust
pub fn hover_text_for_def(
    def_id: DefId,
    def_map: &DefMap,
    interner: &TyInterner,
    type_env: &writ_compiler::check::env::TypeEnv,
    source: &str,
    ast: &TypedAst,
) -> String {
    let entry = def_map.get_entry(def_id);
    match entry.kind {
        DefKind::Fn | DefKind::ExternFn => {
            // Show fn signature (fixes Bug 4: main shows void)
            if let Some(sig) = type_env.fn_sigs.get(&def_id) {
                let sig_text = format_fn_sig_hover(sig, def_map, interner);
                // Append doc comment
                if let Some(doc) = extract_doc_comment(source, entry.span.start) {
                    format!("{}\n\n{}", sig_text, doc)
                } else {
                    sig_text
                }
            } else {
                format!("```writ\nfn {}\n```", entry.name)
            }
        }
        DefKind::Enum => {
            // Show enum name with variants (fixes Bug 5: QuestStatus shows void)
            if let Some(variants) = type_env.enum_variants.get(&def_id) {
                let names: Vec<&str> = variants.iter().map(|v| v.name.as_str()).collect();
                format!("```writ\nenum {} {{ {} }}\n```", entry.name, names.join(", "))
            } else {
                format!("```writ\nenum {}\n```", entry.name)
            }
        }
        DefKind::Const => {
            // Show const with value (fixes Bug 7: MAX_QUESTS shows int without 10)
            let ty = type_env.const_types.get(&def_id)
                .map(|t| interner.display_named(*t, def_map))
                .unwrap_or_else(|| "?".to_string());
            // Extract value source text from the const's value span
            let value_text = ast.decls.iter().find_map(|decl| {
                if let TypedDecl::Const { def_id: did, value, .. } = decl {
                    if *did == def_id {
                        let span = value.span();
                        source.get(span.start..span.end)
                    } else { None }
                } else { None }
            });
            if let Some(val) = value_text {
                format!("```writ\n{}: {} = {}\n```", entry.name, ty, val.trim())
            } else {
                format!("```writ\n{}: {}\n```", entry.name, ty)
            }
        }
        DefKind::Struct | DefKind::ExternStruct => {
            format!("```writ\nstruct {}\n```", entry.name)
        }
        DefKind::Class | DefKind::ExternClass => {
            format!("```writ\nclass {}\n```", entry.name)
        }
        DefKind::Entity => {
            format!("```writ\nentity {}\n```", entry.name)
        }
        DefKind::Global => {
            let ty = type_env.global_types.get(&def_id)
                .map(|t| interner.display_named(*t, def_map))
                .unwrap_or_else(|| "?".to_string());
            format!("```writ\nglobal {}: {}\n```", entry.name, ty)
        }
        _ => String::new(),
    }
}
```

pub fn extract_doc_comment(source: &str, decl_span_start: usize) -> Option<String> {
    let before = &source[..decl_span_start];
    let mut lines: Vec<&str> = Vec::new();
    for line in before.lines().rev() {
        let trimmed = line.trim();
        if trimmed.starts_with("//") {
            lines.push(trimmed.trim_start_matches('/').trim());
        } else if trimmed.is_empty() {
            if !lines.is_empty() { break; } // blank line after comment block = done
            continue; // skip blank lines before finding first comment
        } else {
            break;
        }
    }
    if lines.is_empty() { return None; }
    lines.reverse();
    Some(lines.join("\n"))
}

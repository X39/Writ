**1b. Add `format_stacktrace()` method on `CrashInfo`:**

```rust
impl CrashInfo {
    /// Format the crash as a human-readable string with stack trace.
    ///
    /// Output format:
    /// ```text
    /// Runtime crash: unwrap called on None
    ///
    /// Stack trace:
    ///   at crash_here (line 3, col 5)
    ///   at main (line 7, col 3)
    /// ```
    ///
    /// If the stack trace is empty, returns just the crash message line.
    pub fn format_stacktrace(&self) -> String {
        let mut out = format!("Runtime crash: {}", self.message);
        if !self.stack_trace.is_empty() {
            out.push_str("\n\nStack trace:");
            for frame in &self.stack_trace {
                if frame.line > 0 {
                    out.push_str(&format!(
                        "\n  at {} (line {}, col {})",
                        if frame.method_name.is_empty() {
                            format!("method_{}", frame.method_idx)
                        } else {
                            frame.method_name.clone()
                        },
                        frame.line,
                        frame.column,
                    ));
                } else {
                    out.push_str(&format!(
                        "\n  at {}",
                        if frame.method_name.is_empty() {
                            format!("method_{}", frame.method_idx)
                        } else {
                            frame.method_name.clone()
                        },
                    ));
                }
            }
        }
        out
    }
}

Replace the existing StackFrame construction (around lines 606-618) with:

```rust
let crash_info = crate::error::CrashInfo {
    message: msg.clone(),
    stack_trace: task
        .call_stack
        .iter()
        .rev()
        .map(|f| {
            let loaded = &modules[current_module_idx];

            // Resolve method name from string heap
            let method_name = loaded.module.method_defs
                .get(f.method_idx)
                .and_then(|def| writ_module::heap::read_string(
                    &loaded.module.string_heap, def.name
                ).ok())
                .map(|s| s.to_string())
                .unwrap_or_default();

            // Convert instruction-index PC to byte-offset PC
            let byte_pc = loaded.byte_offsets
                .get(f.method_idx)
                .and_then(|offsets| offsets.get(f.pc))
                .copied()
                .unwrap_or(0);

            // Resolve source location: find largest span.pc <= byte_pc
            let (line, column) = loaded.module.method_bodies
                .get(f.method_idx)
                .and_then(|body| {
                    body.source_spans
                        .iter()
                        .filter(|span| span.pc <= byte_pc)
                        .max_by_key(|span| span.pc)
                        .map(|span| (span.line, span.column as u32))
                })
                .unwrap_or((0, 0));

            crate::error::StackFrame {
                method_idx: f.method_idx,
                method_name,
                pc: f.pc,
                line,
                column,
                registers: f.registers.clone(),
            }
        })
        .collect(),
};
```

**2c. Add `line_col_to_offset` helper:**
```rust
fn line_col_to_offset(source: &str, line: u32, col: u32) -> usize {
    let target_line = line.saturating_sub(1) as usize;
    let mut current_line = 0usize;
    for (idx, ch) in source.char_indices() {
        if current_line == target_line {
            let col_offset = col.saturating_sub(1) as usize;
            let remaining = &source[idx..];
            return remaining.char_indices()
                .nth(col_offset)
                .map(|(i, _)| idx + i)
                .unwrap_or(idx);
        }
        if ch == '\n' {
            current_line += 1;
        }
    }
    0
}
```
